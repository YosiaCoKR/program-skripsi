holdout = TimeSeriesSplit(n_splits=2, test_size=60)
forecasters_eval = {}
eval_rows = []

for col in comodity_cols:
    residual_all = df_cleaned[f"{col}_residual"]
    train_idx, test_idx = list(holdout.split(residual_all))[-1]
    y_train = residual_all.iloc[train_idx]
    y_test = residual_all.iloc[test_idx]
    forecasters_eval[col] = {}

    for h in horizons:
        forecaster_eval = ForecasterDirect(
            estimator=lgb.LGBMRegressor(**best_params_for_GA[col][h]["params"]),
            steps=h,
            lags=[1, 7, 30],
            window_features=window_features,
            calendar_features=calendar_features,
        )
        forecaster_eval.fit(y=y_train)
        forecasters_eval[col][h] = forecaster_eval

        residual_pred = np.asarray(forecaster_eval.predict(steps=h)).ravel()[:h]
        idx = y_test.index[:h]
        harga_aktual = df_cleaned.loc[idx, col].to_numpy()
        harga_pred = (
            df_cleaned.loc[idx, f"{col}_trend"].to_numpy()
            + df_cleaned.loc[idx, f"{col}_seasonal_weekly"].to_numpy()
            + df_cleaned.loc[idx, f"{col}_seasonal_yearly"].to_numpy()
            + residual_pred
        )

        eval_rows.append(
            {
                "Komoditas": col,
                "Horizon": h,
                "MAE": mean_absolute_error(harga_aktual, harga_pred),
                "RMSE": root_mean_squared_error(harga_aktual, harga_pred),
                "MAPE (%)": mean_absolute_percentage_error(harga_aktual, harga_pred)
                * 100,
            }
        )

df_eval = pd.DataFrame(eval_rows)
